// 1. Обработка ошибок
// --------------------
//
// Напиши функцию:
//
// async function divide(a, b) {}
//
// Если b === 0,
// выбрось ошибку.
//
// Вызови функцию через try/catch.

// =========================================
// 2. Загрузка пользователя
// =========================================
//
// Создай функцию:
//
// function getUser()
//
// Она должна вернуть Promise,
// который через 1 секунду вернёт:
//
// {
// id: 1,
// name: 'Vika'
// }
//
// Затем создай:
//
// async function showUser()
//
// Используй await,
// получи пользователя
// и выведи его в консоль.

// =========================================
// 3. Цепочка запросов
// =========================================
//
// Создай три функции:
//
// getUser()
// getPosts(userId)
// getComments(postId)
//
// Все должны возвращать Promise
// через setTimeout.
//
// Например:
//
// getUser()
// ->
// { id: 1, name: 'Vika' }
//
// getPosts(1)
// ->
// [
// { id: 10, title: 'Hello' }
// ]
//
// getComments(10)
// ->
// [
// 'Nice!',
// 'Cool!'
// ]
//
// Используй:
//
// - async/await
// - один try/catch
//
// Ожидаемый вывод:
//
// User: Vika
// Post: Hello
// Comments:
// - Nice!
// - Cool!

// =========================================
// 4. finally
// =========================================
//
// Напиши:
//
// async function loadData() {}
//
// Внутри:
//
// console.log('Loading...');
//
// Затем:
//
// либо верни данные,
// либо выбрось ошибку
// (на выбор).
//
// Но в finally
// обязательно вывести:
//
// Loading finished

// =========================================
// ОГРАНИЧЕНИЯ
// =========================================
//
// ✔ Только async/await.
// ✔ Только try/catch/finally.
// ✘ Не использовать then().

// =========================================
// 5. Promise.all()
// =========================================
//
// Создай функции:
//
// getUser()
// getPosts()
// getSettings()
//
// Каждая должна возвращать Promise
// через setTimeout.
//
// Затем в async функции:
//
// - запусти все три одновременно
// - дождись их через Promise.all()
// - выведи результаты в консоль.
//
// Ожидаемый вывод:
//
// { id: 1, name: 'Vika' }
// [ 'Post 1', 'Post 2' ]
// { theme: 'dark' }
